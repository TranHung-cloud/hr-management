BEGIN;

-- 1. ROLE
CREATE TABLE role (
    role_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    role_name VARCHAR(100) NOT NULL UNIQUE
);

-- 2. DEPARTMENT
CREATE TABLE department (
    dep_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    dep_name VARCHAR(150) NOT NULL UNIQUE
);

-- 3. SKILL
CREATE TABLE skill (
    skill_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    skill_name VARCHAR(100) NOT NULL UNIQUE
);

-- 4. DEGREE TYPE
CREATE TABLE degree_type (
    dt_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    dt_name VARCHAR(100) NOT NULL UNIQUE
);

-- 5. CAMPAIGN
CREATE TABLE campaign (
    cp_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    cp_medium VARCHAR(100),
    cp_source VARCHAR(255)
);

-- 6. INTERVIEW STAGE
CREATE TABLE interview_stage (
    is_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    is_name VARCHAR(100) NOT NULL UNIQUE,
    is_order INT NOT NULL UNIQUE,
    CHECK (is_order > 0)
);

-- 7. INTERVIEW METHOD
CREATE TABLE interview_method (
    im_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    im_name VARCHAR(100) NOT NULL UNIQUE
);

-- 8. ACCOUNT
CREATE TABLE account (
    acc_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    acc_name VARCHAR(100),
    acc_email VARCHAR(255) NOT NULL UNIQUE,
    acc_password VARCHAR(255) NOT NULL,
    acc_gender VARCHAR(20),
    acc_dob DATE,
    acc_phone VARCHAR(20),

    role_id UUID REFERENCES role(role_id),
    dep_id UUID REFERENCES department(dep_id)
);

-- 9. JOB DESCRIPTION
CREATE TABLE job_description (
    job_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    job_title VARCHAR(200) NOT NULL,
    job_desc TEXT,

    salary_min INT,
    salary_max INT,

    job_status VARCHAR(50) NOT NULL DEFAULT 'DRAFT',
    job_target VARCHAR(255),
    job_major VARCHAR(150),

    dep_id UUID REFERENCES department(dep_id),
    dt_id UUID REFERENCES degree_type(dt_id),
    recruiter_id UUID REFERENCES account(acc_id),

    CHECK (job_status IN ('DRAFT', 'PUBLISHED')),
    CHECK (salary_min IS NULL OR salary_min >= 0),
    CHECK (salary_max IS NULL OR salary_max >= 0),

    CHECK (
        salary_min IS NULL
        OR salary_max IS NULL
        OR salary_min <= salary_max
    )
);

-- 10. CANDIDATE
CREATE TABLE candidate (
    cand_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    cand_note TEXT,
    cand_created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    cand_expected_sal INT,
    cand_proposal_sal INT,

    cand_cv_path VARCHAR(500),
    cand_exp_year INT,
    cand_major VARCHAR(150),

    acc_id UUID UNIQUE REFERENCES account(acc_id),
    dt_id UUID REFERENCES degree_type(dt_id),

    CHECK (
        cand_expected_sal IS NULL
        OR cand_expected_sal >= 0
    ),

    CHECK (
        cand_proposal_sal IS NULL
        OR cand_proposal_sal >= 0
    ),

    CHECK (
        cand_exp_year IS NULL
        OR cand_exp_year >= 0
    )
);

-- 11. SKILL CANDIDATE
CREATE TABLE skill_candidate (
    cand_id UUID NOT NULL
        REFERENCES candidate(cand_id) ON DELETE CASCADE,

    skill_id UUID NOT NULL
        REFERENCES skill(skill_id) ON DELETE CASCADE,

    PRIMARY KEY (cand_id, skill_id)
);

-- 12. SKILL JOB DESCRIPTION
CREATE TABLE skill_job_description (
    job_id UUID NOT NULL
        REFERENCES job_description(job_id) ON DELETE CASCADE,

    skill_id UUID NOT NULL
        REFERENCES skill(skill_id) ON DELETE CASCADE,

    PRIMARY KEY (job_id, skill_id)
);

-- 13. APPLICATION
CREATE TABLE application (
    status VARCHAR(50) NOT NULL DEFAULT 'SUBMITTED',

    cand_id UUID NOT NULL
        REFERENCES candidate(cand_id),

    job_id UUID NOT NULL
        REFERENCES job_description(job_id),

    CHECK (
        status IN (
            'SUBMITTED',
            'PENDING',
            'ACCEPTED',
            'REJECTED'
        )
    ),

    PRIMARY KEY (cand_id, job_id)
);

-- 14. INTERVIEW PROCESS
CREATE TABLE interview_process (
    cand_id UUID NOT NULL,
    job_id UUID NOT NULL,

    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    status VARCHAR(50) NOT NULL DEFAULT 'UNSCHEDULED',
    scheduled_at TIMESTAMP,
    feedback TEXT,

    acc_id UUID REFERENCES account(acc_id),
    is_id UUID NOT NULL REFERENCES interview_stage(is_id),
    im_id UUID REFERENCES interview_method(im_id),

    PRIMARY KEY (cand_id, job_id, is_id),

    FOREIGN KEY (cand_id, job_id)
        REFERENCES application(cand_id, job_id)
        ON DELETE CASCADE,

    CHECK (
        status IN (
            'SCHEDULED',
            'UNSCHEDULED',
            'COMPLETED',
            'REJECTED'
        )
    )
);

-- 15. CAMPAIGN JOB DESCRIPTION
CREATE TABLE campaign_job_description (
    cp_id UUID NOT NULL
        REFERENCES campaign(cp_id) ON DELETE CASCADE,

    job_id UUID NOT NULL
        REFERENCES job_description(job_id) ON DELETE CASCADE,

    PRIMARY KEY (cp_id, job_id)
);
